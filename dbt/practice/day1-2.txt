demonstrate rel betn tables
here we tell sql that if there's not a student in student table with an id dont add in result table thats how foreign key is use
rank is keyword in modern sql so use stud_rank
create table student_result (
	stud_id int primary key,
	marks int,
	stud_rank int,
	constraint fk_res_stud
		foreign key (stud_id) references student(id)
);


insert vals in res

insert into student_result (stud_id,marks,stud_rank)

(2,97,3),
(3,99,1),

select specific student

select
	student.id,
	student.name,
	student.course,
	student_result.marks,
	student_result.stud_rank
from student, student_result
where student.id = student_result.stud_id
and student.name = 'satyam';
