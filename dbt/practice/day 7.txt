create procedure class_summary()
begin
	declare v_avg decimal(5,2);
	declare v_max decimal(5,2) default 0;
	select avg(marks), max(marks) into v_avg,v_max from student;
	select v_avg as average, v_max as highest, v_max - v_avg as gap;
end//
delimiter ;


call class_summary();

create procedure stud_grade(in p_id int)
begin
	declare v_marks decimal(5,2);
	declare v_grade char(1);
	select marks into v_marks from student where student_id = p_id;
	if v_marks >= 90 then set v_grade = 'a';
	elseif v_marks >=80  then set v_grade = 'b';
	elseif v_marks >=70 then set v_grade = 'c';
	else set v_grade = 'f';
	end if;
	select p_id as student_id,v_marks as marks, v_grade as grade;
end //

delimiter //
create procedure factorial_of(in p_n int)
begin
	declare i int default 1;
	declare f bigint default 1;
	while i <= p_n do
		set f = f * i;
		set i = i + 1;
	end while;
	select p_n as n, f as factorial;
end //

create procedure sum_odd(in p_n int, out p_sum int)
begin
	declare i int default 0;
	set p_sum = 0;
	odd_loop : loop
		set i = i+1;
		if i > p_n then leave odd_loop; end if;
		if i mod 2 = 0 then iterate odd_loop;end if;
		set p_sum = p_sum + i;
	end loop odd_loop;
end //

create procedure count_up(in p_n int)
begin
	declare i int default 0;
	repeat
		set I = i+1;
	until I >= p_n end repeat ;
	select i as ended_at;
end //

create function grade(p_marks decimal(5,2)) returns char(1) deterministic
begin
	declare g char(1);
	if p_marks >= 90 then set g = 'a';
	elseif p_marks >=80 then set g = 'b';
	elseif p_marks >= 70 then set g ='c';
	else set g = 'f';
	end if;
	return g;
end //
