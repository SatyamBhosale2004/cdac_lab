create table passport_registration (id int primary key auto_increment, name varchar(20));

create table passport_application (
	app_no int primary key,
	status enum('rejected','successful','submitted'),
	reg_id int,
	app_date date,
	constraint fk_app_reg_id foreign key (reg_id) references passport_registration  	(id) on delete cascade on update cascade
);
